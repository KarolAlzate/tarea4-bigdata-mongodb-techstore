// Filtros con operadores lógicos y de comparación ($in, $gt, $lte, $gte)
db.productos.find({
  categoria: { $in: ["Computadores", "Monitores"] },
  precio: { $gt: 500000, $lte: 1500000 },
  stock: { $gte: 10 }
});

// Operación Join ($lookup) entre 'ventas' y 'productos'
db.ventas.aggregate([
  {
    $lookup: {
      from: "productos",
      localField: "producto_id",
      foreignField: "producto_id",
      as: "detalle_producto"
    }
  },
  { $unwind: "$detalle_producto" },
  {
    $project: {
      _id: 0,
      venta_id: 1,
      cliente: 1,
      ciudad: 1,
      producto: "$detalle_producto.nombre",
      precio_unitario: "$detalle_producto.precio",
      cantidad: 1,
      total_venta: { $multiply: ["$cantidad", "$detalle_producto.precio"] }
    }
  }
]);
// Consultas de agregación: contar, sumar, promediar y máximo por categoría
db.productos.aggregate([
  {
    $group: {
      _id: "$categoria",
      total_productos: { $sum: 1 },
      suma_stock_total: { $sum: "$stock" },
      precio_promedio: { $avg: "$precio" },
      precio_maximo: { $max: "$precio" }
    }
  },
  { $sort: { precio_promedio: -1 } }
]);
